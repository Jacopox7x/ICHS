# 1.5 DETECT EVENT TYPE (professional / leisure)
# ============================================

def detect_event_type(facts):
    professional_keywords = [
        "conferenza", "summit", "workshop", "formazione", "business",
        "networking", "professionisti", "azienda", "corporate", "industry",
        "tecnologia", "innovazione", "startup", "investitori", "manager",
        "ceo", "cto", "policy maker", "sviluppo", "digitale", "marketing"
    ]
    leisure_keywords = [
        "festa", "compleanno", "matrimonio", "cena", "amici",
        "famiglia", "vacanza", "spiaggia", "divertimento", "party",
        "concerto", "spettacolo", "culturale", "arte", "sport",
        "runner", "maratona", "turistico", "relax", "gioia"
    ]
    texts = []
    for key in ['event_name', 'target', 'objective', 'location']:
        value = facts.get(key, '')
        if value and value != "DA DEFINIRE":
            texts.append(value.lower())
    full_text = " ".join(texts)
    professional_score = sum(1 for kw in professional_keywords if kw in full_text)
    leisure_score = sum(1 for kw in leisure_keywords if kw in full_text)
    if professional_score > leisure_score:
        return "professionale"
    elif leisure_score > professional_score:
        return "leisure"
    else:
        target = facts.get('target', '').lower()
        if any(kw in target for kw in ["amici", "famiglia", "divertimento"]):
            return "leisure"
        elif any(kw in target for kw in ["professionisti", "manager", "ceo"]):
            return "professionale"
        return "professionale"

def generate_creative_with_qwen(facts):
    event = facts.get('event_name', 'Evento')
    date = facts.get('date', 'data da definire')
    location = facts.get('location', 'location da definire')
    
    prompt = f"""Genera 3 idee creative per promuovere l'evento "{event}".
Data: {date}
Location: {location}

Formato:
IDEA: [titolo]
CONCEPT: [descrizione]
PERCHE FUNZIONA: [spiegazione]
RISCHIO: [criticita]"""
    
    try:
        payload = {
            "model": MODEL,
            "messages": [{"role": "user", "content": prompt}],
            "stream": False,
            "think": False,
            "options": {"temperature": 0.8, "num_predict": 1500}
        }
        req = urllib.request.Request(
            "http://localhost:11434/api/chat",
            data=json.dumps(payload).encode('utf-8'),
            headers={'Content-Type': 'application/json'}
        )
        with urllib.request.urlopen(req, timeout=180) as r:
            d = json.loads(r.read().decode())
            resp = d.get("message", {}).get("content", "").strip()
        return resp if len(resp) >= 10 else None
    except Exception as e:
        print(f"  [ERR] Qwen: {e}")
        return None

    match = re.search(r'(?:Obiettivo|Scopo|obiettivo|scopo):?\s*([^.,;]+)', text, re.IGNORECASE)
    if match:

    keywords = ['attirare', 'promuovere', 'formare', 'sensibilizzare', 'creare', 'raccogliere', 'divertire']
    for kw in keywords:

from typing import Dict
import json
import os

# Database città (da espandere)
CITY_DATA = {
    'Roma': {'population': 2_873_000, 'region': 'Lazio'},
    'Milano': {'population': 1_396_000, 'region': 'Lombardia'},
    'Torino': {'population': 848_000, 'region': 'Piemonte'},

    'Genova': {'population': 558_000, 'region': 'Liguria'},
    'Palermo': {'population': 630_000, 'region': 'Sicilia'},
}

# Penetrazione social per settore (dati ISTAT + benchmark)
SOCIAL_PENETRATION = {
    'professionale': {
        'linkedin': 0.35,  # 35% dei professionisti su LinkedIn
        'instagram': 0.20,
        'twitter': 0.15,
    },
    'leisure': {
        'linkedin': 0.10,
        'instagram': 0.55,
        'twitter': 0.25,
    },
    'tecnico': {
        'linkedin': 0.45,
        'instagram': 0.15,
        'twitter': 0.30,
    }
}

# Settori per tipo di evento
SECTOR_KEYWORDS = {
    'tecnico': ['tecnico', 'industria', 'ferro', 'acciaio', 'macchinari', 'produzione'],
    'professionale': ['manager', 'ceo', 'commercialista', 'consulente', 'avvocato'],
    'leisure': ['festa', 'divertimento', 'amici', 'party', 'concerto'],
}

def detect_sector(facts: Dict, text: str) -> str:
    """Determina il settore in base a target e testo"""
    text_lower = (text + ' ' + str(facts)).lower()
    for sector, keywords in SECTOR_KEYWORDS.items():
        if any(kw in text_lower for kw in keywords):
            return sector
    # Default: professionale per eventi B2B, leisure per eventi B2C
    if any(kw in text_lower for kw in ['festa', 'divertimento', 'amici']):
        return 'leisure'
    return 'professionale'

def get_city_population(location: str) -> int:
    """Ottiene popolazione della città dal database"""
    for city, data in CITY_DATA.items():
        if city.lower() in location.lower():
            return data['population']
    return 500_000  # Default per città non mappate

def estimate_reach(facts: Dict, text: str) -> Dict[str, int]:
    """
    Stima reach basata su:
    1. Popolazione città
    2. Penetrazione social per settore
    3. Fattore di conversione per tipo evento
    """
    
    location = facts.get('location', '')
    sector = detect_sector(facts, text)
    population = get_city_population(location)
    
    # Fattore di conversione: quanta parte della popolazione è target
    # Per eventi locali: 1-5% della popolazione
    # Per eventi regionali: 5-15%
    # Per eventi nazionali: 15-30%
    conversion_factor = 0.02  # Default: 2% della popolazione (locale)
    
    # Aggiusta per settore
    if sector == 'tecnico':
        conversion_factor = 0.03  # Più alto per settori di nicchia
    elif sector == 'professionale':
        conversion_factor = 0.025
    elif sector == 'leisure':
        conversion_factor = 0.05  # Più alto per eventi leisure
    
    # Calcola target potenziale
    potential_target = int(population * conversion_factor)
    
    # Reach per piattaforma
    penetration = SOCIAL_PENETRATION.get(sector, SOCIAL_PENETRATION['professionale'])
    
    reach = {}
    for platform, pct in penetration.items():
        reach[platform] = int(potential_target * pct * 1.5)  # 1.5x per social sharing
    
    # Assicura che il reach sia almeno un minimo credibile
    for platform in ['linkedin', 'instagram', 'twitter']:
        if reach.get(platform, 0) < 500:
            reach[platform] = 500
    
    return reach

def get_reach_report(facts: Dict, text: str) -> Dict:
    """Genera report reach con spiegazione"""
    reach = estimate_reach(facts, text)
    total = sum(reach.values())
    
    location = facts.get('location', '')
    sector = detect_sector(facts, text)
    
    return {
        'reach': reach,
        'total': total,
        'location': location,
        'sector': sector,
        'population': get_city_population(location),
        'conversion_rate': f"{0.02 if sector == 'professionale' else 0.05:.1%}",
        'explanation': f"Basato su popolazione di {location} ({get_city_population(location):,} abitanti) e settore {sector}."
    }

import json
import re
import subprocess
from typing import List, Dict

class HashtagGenerator:
    def __init__(self):
        self.pools = {
            'professionale': ['#Business', '#Leadership', '#Strategy', '#Growth', '#Success'],
            'leisure': ['#Party', '#Divertimento', '#Amici', '#Weekend', '#Vibes'],
            'generale': ['#Evento', '#Innovazione', '#Networking', '#Community', '#Experience']
        }
    
    def extract_from_brief(self, text: str) -> List[str]:
        return re.findall(r'#\w+', text)
    
    def generate_with_qwen(self, facts: Dict[str, str]) -> List[str]:
        """Genera hashtag NATURALI con Qwen - NO giochi di parole forzati"""
        prompt = f"""Genera 15 hashtag per questo evento.

REGOLE IMPORTANTI:
1. Usa hashtag che le persone userebbero DAVVERO
2. NIENTE giochi di parole forzati o creativi a tutti i costi
3. Usa combinazioni di parole comuni e naturali
4. Mescola italiano e inglese in modo spontaneo
5. Ogni hashtag deve sembrare "ovvio" in retrospettiva

Esempi di hashtag NATURALI (non forzati):
#BirraArtigianale #BolognaEvent #CulturaDellaBirra #BirraBologna

Evento: {facts.get('event_name')}
Luogo: {facts.get('location')}
Target: {facts.get('target')}
Obiettivo: {facts.get('objective')}

Rispondi SOLO con JSON: {{"hashtags": ["#tag1", ...]}}"""
        
        try:
            payload = json.dumps({
                "model": "qwen3.5:4b",
                "messages": [{"role": "user", "content": prompt}],
                "stream": False,
                "options": {"temperature": 0.3}  # Più bassa per meno creatività forzata
            })
            
            result = subprocess.run(
                ["curl", "-s", "-X", "POST", "http://localhost:11434/api/chat",
                 "-H", "Content-Type: application/json", "-d", payload],
                capture_output=True, text=True, timeout=90
            )
            
            if result.returncode == 0 and result.stdout:
                data = json.loads(result.stdout)
                content = data.get("message", {}).get("content", "")
                json_match = re.search(r'\{[^{}]*\}', content, re.DOTALL)
                if json_match:
                    parsed = json.loads(json_match.group(0))
                    return parsed.get("hashtags", [])
        except:
            pass
        return []
    
    def generate_fallback(self, facts: Dict[str, str]) -> List[str]:
        hashtags = []
        event = facts.get('event_name', '')
        if event and event != 'DA DEFINIRE':
            words = re.findall(r'[A-Za-z]+', event)
            if words:
                hashtags.append('#' + ''.join(words[:3]))
                if len(words) >= 2:
                    hashtags.append('#' + ''.join(words[:2]))
        location = facts.get('location', '')
        if location and location != 'DA DEFINIRE':
            hashtags.append(f'#{location}')
        target = facts.get('target', '')
        if target and target != 'DA DEFINIRE':
            for word in target.split():
                if len(word) > 3:
                    hashtags.append('#' + word.capitalize())
        objective = facts.get('objective', '')
        if objective and objective != 'DA DEFINIRE':
            words = re.findall(r'[A-Za-z]{4,}', objective)
            for word in words[:2]:
                if word.lower() not in ['promuovere', 'cultura', 'evento']:
                    hashtags.append('#' + word.capitalize())
        text = str(facts).lower()
        if any(kw in text for kw in ['festa', 'party', 'divertimento']):
            hashtags.extend(self.pools['leisure'])
        else:
            hashtags.extend(self.pools['professionale'])
        hashtags.extend(self.pools['generale'])
        seen = set()
        unique = []
        for tag in hashtags:
            if tag not in seen:
                seen.add(tag)
                unique.append(tag)
        return unique[:15]
    
    def generate(self, text: str, facts: Dict[str, str]) -> List[str]:
        # 1. Hashtag dal brief
        hashtags = self.extract_from_brief(text)
        if hashtags:
            print("🔍 Hashtag dal brief (priorità 1)")
            return hashtags[:15]
        
        # 2. Qwen (temperatura 0.3, no creatività forzata)
        print("🎨 Generazione hashtag naturali con Qwen... (fino a 90s)")
        hashtags = self.generate_with_qwen(facts)
        if hashtags:
            print(f"✅ {len(hashtags)} hashtag generati")
            return hashtags[:15]
        
        # 3. Fallback
        print("🔍 Hashtag fallback (priorità 3)")
        return self.generate_fallback(facts)[:15]

def generate_hashtags(text: str, facts: Dict[str, str]) -> List[str]:
    return HashtagGenerator().generate(text, facts)

import duckdb

from typing import Dict, List, Optional

class OutputManager:
    def __init__(self, db_path: str = "icrawler_outputs.duckdb"):
        self.db_path = db_path
        self.conn = duckdb.connect(db_path)
        self.init_tables()
    
    def init_tables(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS plans (
                plan_id VARCHAR PRIMARY KEY,
                brief TEXT,
                facts JSON,
                hashtags JSON,
                creative_lab JSON,
                full_plan TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS hashtag_cache (
                cache_key VARCHAR PRIMARY KEY,
                hashtags JSON,
                category VARCHAR,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS plan_analytics (
                plan_id VARCHAR,
                platform VARCHAR,
                reach_estimate INTEGER,
                engagement_score FLOAT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
    
    def save_plan(self, plan_id: str, brief: str, facts: Dict, hashtags: List, creative_lab: Dict, full_plan: str) -> bool:
        try:
            self.conn.execute("""
                INSERT OR REPLACE INTO plans (plan_id, brief, facts, hashtags, creative_lab, full_plan)
                VALUES (?, ?, ?, ?, ?, ?)
            """, (plan_id, brief, json.dumps(facts), json.dumps(hashtags), json.dumps(creative_lab), full_plan))
            self.conn.commit()
            return True
        except Exception as e:
            print(f"❌ Errore salvataggio piano: {e}")
            return False
    
    def get_plan(self, plan_id: str) -> Optional[Dict]:
        result = self.conn.execute("""
            SELECT plan_id, brief, facts, hashtags, creative_lab, full_plan, created_at
            FROM plans WHERE plan_id = ?
        """, (plan_id,)).fetchone()
        if result:
            return {
                'plan_id': result[0],
                'brief': result[1],
                'facts': json.loads(result[2]),
                'hashtags': json.loads(result[3]),
                'creative_lab': json.loads(result[4]),
                'full_plan': result[5],
                'created_at': result[6]
            }
        return None
    
    def get_stats(self) -> Dict:
        total_plans = self.conn.execute("SELECT COUNT(*) FROM plans").fetchone()[0]
        total_hashtags_cached = self.conn.execute("SELECT COUNT(*) FROM hashtag_cache").fetchone()[0]

            'total_plans': total_plans,
            'total_hashtags_cached': total_hashtags_cached,
            'total_analytics': total_analytics
        }
    
    def close(self):
        self.conn.close()

CITIES = {
    'Roma', 'Milano', 'Torino', 'Bologna', 'Firenze', 'Rovigo',
    'Venezia', 'Napoli', 'Genova', 'Palermo', 'Bari', 'Verona'
}

MONTHS = {
    'gennaio', 'febbraio', 'marzo', 'aprile', 'maggio', 'giugno',
    'luglio', 'agosto', 'settembre', 'ottobre', 'novembre', 'dicembre',
    'gen', 'feb', 'mar', 'apr', 'mag', 'giu', 'lug', 'ago', 'set', 'ott', 'nov', 'dic'
}

def extract_city(text: str) -> str:
    for city in CITIES:
        if city.lower() in text.lower():
            return city
    match = re.search(r'a\s+([A-Z][a-z]+)', text)
    if match:
        return match.group(1)
    return 'DA DEFINIRE'

def extract_date(text: str) -> str:
    # Data completa con anno
    match = re.search(r'(\d{1,2}\s+([A-Za-z]+)\s+\d{4})', text, re.IGNORECASE)
    if match and match.group(2).lower() in MONTHS:
        return match.group(1).strip()
    # Intervallo
    match = re.search(r'(\d{1,2}[-–]\d{1,2}\s+([A-Za-z]+))', text, re.IGNORECASE)
    if match and match.group(2).lower() in MONTHS:
        return match.group(1).strip()
    # "il" + giorno + mese
    match = re.search(r'il\s+(\d{1,2}\s+([A-Za-z]+))', text, re.IGNORECASE)
    if match and match.group(2).lower() in MONTHS:
        return match.group(1).strip()
    # Giorno + mese
    match = re.search(r'(\d{1,2}\s+([A-Za-z]+))(?=\s|\.|,|$)', text, re.IGNORECASE)
    if match and match.group(2).lower() in MONTHS:
        return match.group(1).strip()
    # Solo anno
    match = re.search(r'(\d{4})', text)
    if match:
        year = int(match.group(1))
        if 2000 <= year <= 2099:
            return match.group(1)
    return 'DA DEFINIRE'

def extract_objective(text: str) -> str:
    match = re.search(r'(?:Obiettivo|Scopo):?\s*([^.,;]+)', text, re.IGNORECASE)
    if match:
        return match.group(1).strip()
    return 'DA DEFINIRE'

def clean_event_name(event_name: str, full_text: str) -> str:
    if not event_name or event_name == 'DA DEFINIRE':
        return 'DA DEFINIRE'
    
    # 1. Rimuovi città (es. " a Milano", ", Milano")
    for city in CITIES:
        event_name = re.sub(rf'\s*a\s*{city}\b', '', event_name, flags=re.IGNORECASE)
        event_name = re.sub(rf'\s*,\s*{city}\b', '', event_name, flags=re.IGNORECASE)
    
    # 2. Rimuovi la data estratta
    date = extract_date(full_text)
    if date != 'DA DEFINIRE':
        # Rimuovi date in vari formati
        event_name = re.sub(rf'\s*{re.escape(date)}\s*', ' ', event_name, flags=re.IGNORECASE)
        # Rimuovi pattern "2026" generici
        event_name = re.sub(r'\s*\d{4}\s*', ' ', event_name)
    
    # 3. Rimuovi "il" e "a" residui
    event_name = re.sub(r'\s*il\s*', ' ', event_name)
    event_name = re.sub(r'\s*a\s*$', '', event_name)
    event_name = re.sub(r'^\s*a\s*', '', event_name)
    
    # 4. Rimuovi residui di date mal tagliate (es. "20 M ano", "26 il")
    #    Pattern: numero + parola + parola (che non siano mesi)
    event_name = re.sub(r'\d+\s+[A-Za-z]+\s+[A-Za-z]+', ' ', event_name)
    #    Pattern: numero + parola (che non sia un mese)
    event_name = re.sub(r'\d+\s+[A-Za-z]+', ' ', event_name)
    
    # 5. Rimuovi spazi multipli e virgole finali
    event_name = re.sub(r'\s+', ' ', event_name).strip()
    event_name = event_name.strip().strip(',')
    
    # 6. Se il nome inizia con "DA DEFINIRE", fallback: prendi le prime 2 parole
    if event_name == 'DA DEFINIRE' or len(event_name) < 3:
        # Prendi le prime parole significative (max 3)
        words = full_text.split()
        if words:
            # Prendi fino a 3 parole, escludendo date e città
            candidates = []
            for w in words[:6]:
                # Salta se sembra una data o città
                if w in MONTHS or w in CITIES or re.match(r'\d+', w):
                    continue
                candidates.append(w)
                if len(candidates) >= 3:
                    break
            if candidates:
                return ' '.join(candidates)
    
    return event_name if event_name else 'DA DEFINIRE'

def post_process(facts: Dict[str, str], full_text: str) -> Dict[str, str]:
    result = facts.copy()
    
    result['location'] = extract_city(full_text)
    result['date'] = extract_date(full_text)
    
    if result.get('objective') == 'DA DEFINIRE':
        result['objective'] = extract_objective(full_text)
    
    result['event_name'] = clean_event_name(
        result.get('event_name', ''),
        full_text
    )
    
    target = result.get('target', '')
    for prefix in ['Per ', 'rivolto a ']:
        if target.startswith(prefix):
            target = target[len(prefix):]
    result['target'] = target.strip()
    
    return result

import json
import re
import subprocess
from typing import Dict, Optional

PROMPT_TEMPLATE = """Estrai i seguenti campi dal brief:

- event_name: il nome dell'evento (NON deve contenere la città)
- date: la data
- location: SOLO il nome della città, senza "a" davanti
- target: il pubblico
- objective: l'obiettivo
- hashtag: se presente

REGOLE FERREE:
1. event_name NON può contenere "a Bologna", "a Milano", "a Roma" o qualsiasi città
2. location è SOLO il nome della città (es. "Bologna", non "a Bologna")
3. Se un campo non è presente, usa "DA DEFINIRE"

Brief: {text}

Rispondi SOLO in JSON valido, senza spiegazioni."""

def parse_response(content: str) -> Dict[str, str]:
    result = {}
    
    json_match = re.search(r'\{[^{}]*\}', content, re.DOTALL)
    if json_match:
        try:
            parsed = json.loads(json_match.group(0))
            if isinstance(parsed, dict):
                return parsed
        except:
            pass
    
    patterns = {
        'event_name': r'(?:event_name|nome)[\s:]+["\']?([^"\',\n}]+)',
        'date': r'(?:date|data)[\s:]+["\']?([^"\',\n}]+)',
        'location': r'(?:location|luogo|città)[\s:]+["\']?([^"\',\n}]+)',
        'target': r'(?:target|pubblico)[\s:]+["\']?([^"\',\n}]+)',
        'objective': r'(?:objective|obiettivo)[\s:]+["\']?([^"\',\n}]+)',
        'hashtag': r'(?:hashtag)[\s:]+["\']?([^"\',\n}]+)'
    }
    
    for key, pattern in patterns.items():
        match = re.search(pattern, content, re.IGNORECASE)
        if match:
            value = match.group(1).strip().strip('",')
            if value and value.lower() not in ['null', 'none']:
                result[key] = value
    
    return result

def extract_with_qwen(text: str) -> Optional[Dict[str, str]]:
    prompt = PROMPT_TEMPLATE.format(text=text)
    
    payload = json.dumps({
        "model": "qwen3.5:4b",
        "messages": [{"role": "user", "content": prompt}],
        "stream": False,
        "options": {"temperature": 0.0}
    })
    
    try:
        result = subprocess.run(
            ["curl", "-s", "-X", "POST", "http://localhost:11434/api/chat",
             "-H", "Content-Type: application/json", "-d", payload],
            capture_output=True, text=True, timeout=30
        )
        
        if result.returncode == 0 and result.stdout:
            data = json.loads(result.stdout)
            content = data.get("message", {}).get("content", "")
            parsed = parse_response(content)
            if parsed:
                for field in ['event_name', 'date', 'location', 'target', 'objective', 'hashtag']:
                    if field not in parsed:
                        parsed[field] = 'DA DEFINIRE'
                return parsed
        return None
    except Exception:
        return None

import re
from typing import Dict

def extract_with_regex(text: str) -> Dict[str, str]:
    """Estrae facts con regex (fallback robusto)"""
    
    facts = {
        'event_name': 'DA DEFINIRE',
        'date': 'DA DEFINIRE',
        'location': 'DA DEFINIRE',
        'organizer': 'DA DEFINIRE',
        'target': 'DA DEFINIRE',
        'objective': 'DA DEFINIRE',
        'time': 'DA DEFINIRE',
        'venue': 'DA DEFINIRE',
        'speaker': 'DA DEFINIRE',
        'budget': 'DA DEFINIRE',
        'link': 'DA DEFINIRE',
        'hashtag': 'DA DEFINIRE'
    }
    
    # DATE
    date_match = re.search(r'(\d{1,2}[-–]\d{1,2}\s+[A-Za-z]+)', text)
    if not date_match:
        date_match = re.search(r'(\d{1,2}\s+[A-Za-z]+)', text)
    if date_match:
        facts['date'] = date_match.group(1)
    
    # LOCATION
    loc_match = re.search(r'a\s+([A-Z][a-z]+)(?=\s|\.|,|$)', text)
    if loc_match:
        loc = loc_match.group(1)
        if loc not in ['Birra', 'Cultura', 'Tutta', 'Europa']:
            facts['location'] = loc
    
    # TARGET
    target_match = re.search(r'(?:per|rivolto a)\s+([^.,;]+)', text, re.IGNORECASE)
    if target_match:
        facts['target'] = target_match.group(1).strip()
    
    # OBJECTIVE
    for kw in ['promuovere', 'attirare', 'formare', 'sensibilizzare', 'creare', 'raccogliere', 'divertire', 'networking']:
        if kw in text.lower():
            for s in text.split('.'):
                if kw in s.lower():
                    facts['objective'] = s.strip()
                    break
            if facts['objective'] != 'DA DEFINIRE':
                break
    
    # EVENT NAME
    if ' il ' in text:
        facts['event_name'] = text.split(' il ')[0].strip()
    else:
        facts['event_name'] = text.split('.')[0].strip()
    if facts['date'] in facts['event_name']:
        facts['event_name'] = facts['event_name'].replace(facts['date'], '').replace(',', '').strip()
    
    # HASHTAG
    htag = re.search(r'(#[a-zA-Z0-9_]+)', text)
    if htag:
        facts['hashtag'] = htag.group(1)
    
    return facts

"""
iCRAWLER — Cache per evitare rigenerazioni duplicate
"""

import hashlib
import json
from pathlib import Path

class ICrawlerCache:
    def __init__(self, cache_dir="cache"):
        self.cache_dir = cache_dir
        Path(cache_dir).mkdir(exist_ok=True)
    
    def get_key(self, brief, template):
        content = f"{brief}|{template}"
        return hashlib.md5(content.encode()).hexdigest()
    
    def get(self, key):
        path = Path(self.cache_dir) / f"{key}.json"
        if path.exists():
            with open(path) as f:
                return json.load(f)
        return None
    
    def set(self, key, value):
        path = Path(self.cache_dir) / f"{key}.json"
        with open(path, "w") as f:
            json.dump(value, f)

import requests

def get_hashtags_from_cloud(topic, count=10):
    try:
        url = "https://api.ritekit.com/v1/stats/hashtag-suggestions"
        params = {"text": topic}
        response = requests.get(url, params=params, timeout=3)
        if response.status_code == 200:
            data = response.json()
            hashtags = []
            for item in data.get('data', []):
                if 'hashtag' in item:
                    hashtags.append(item['hashtag'])
            if hashtags:
                return hashtags[:count]
    except:
        pass
    
    base = topic.replace(" ", "").lower()
    return [
        f"#{base}", f"#{base}2026", f"#{base}Event",
        f"#{base}Community", f"#{base}Innovation",
        f"#{base}Life", f"#{base}Daily", f"#{base}World",
        f"#{base}Experience", f"#{base}Vibes"
    ][:count]

        self.conn.row_factory = sqlite3.Row

    def retrieve(self, platform=None):
        context = {}

        cur = self.conn.cursor()

        if platform:
            cur.execute("""
                SELECT * FROM platform_rules
                WHERE platform=?
                ORDER BY priority DESC
            """, (platform,))
            context["platform_rules"] = [dict(x) for x in cur.fetchall()]

            "style_rules",
            "content_rules",
            "hashtag_rules",
            "visual_rules",
            "cta_rules",
            "anti_patterns",
            "validation_rules"
        ]:
            cur.execute(f"""
                SELECT * FROM {table}
                ORDER BY priority DESC
            """)
            context[table] = [dict(x) for x in cur.fetchall()]

        return context

    def close(self):
        self.conn.close()

"""

"""

import json
from datetime import datetime
from pathlib import Path

class ICrawlerLogger:
    def __init__(self, log_file="logs/icrawler.log"):
        self.log_file = log_file
        Path("logs").mkdir(exist_ok=True)
    
    def log(self, event_type, data):
        entry = {
            "timestamp": datetime.now().isoformat(),
            "event": event_type,
            "data": data
        }
        with open(self.log_file, "a") as f:
            f.write(json.dumps(entry) + "\n")
    
    def get_history(self, limit=50):
        if not Path(self.log_file).exists():
            return []
        with open(self.log_file, "r") as f:
            lines = f.readlines()
        return [json.loads(line) for line in lines[-limit:]]

import json
from pathlib import Path
from datetime import datetime, timedelta

class ICrawlerMetrics:
    def __init__(self, metrics_file="metrics/icrawler_metrics.json"):
        self.metrics_file = metrics_file
        Path("metrics").mkdir(exist_ok=True)
        self._init_metrics()
    
    def _init_metrics(self):
        if not Path(self.metrics_file).exists():
            self.data = {
                "total_generations": 0,
                "qwen_success": 0,
                "qwen_failures": 0,
                "fallback_used": 0,
                "avg_characters": 0,
                "avg_time": 0,
                "daily": {}
            }
            self._save()
        else:
            with open(self.metrics_file, "r") as f:
                self.data = json.load(f)
    
    def _save(self):
        with open(self.metrics_file, "w") as f:
            json.dump(self.data, f, indent=2)
    
    def record_generation(self, chars, time_sec, qwen_used):
        self.data["total_generations"] += 1
        if qwen_used:
            self.data["qwen_success"] += 1
        else:
            self.data["fallback_used"] += 1
        
        # Media
        total = self.data["total_generations"]
        self.data["avg_characters"] = ((self.data["avg_characters"] * (total - 1)) + chars) / total
        self.data["avg_time"] = ((self.data["avg_time"] * (total - 1)) + time_sec) / total
        
        # Daily
        day = datetime.now().strftime("%Y-%m-%d")
        if day not in self.data["daily"]:
            self.data["daily"][day] = 0
        self.data["daily"][day] += 1
        
        self._save()
    
    def get_report(self):

"""
iCRAWLER — Validazione automatica dell'output
"""

import re

class OutputValidator:
    def __init__(self, facts):
        self.facts = facts
        self.errors = []
        self.warnings = []
    
    def validate(self, output):
        # 1. Controlla che i fatti siano presenti
        for key, value in self.facts.items():
            if value and value != "DA DEFINIRE" and key != "hashtag":
                if str(value) not in output:
                    self.errors.append(f"Fact '{key}' non trovato nell'output")
        
        # 2. Controlla che non ci siano "DA DEFINIRE" nel core output
        if "DA DEFINIRE" in output and "CREATIVE LAB" in output:
            core = output.split("CREATIVE LAB")[0]
            if "DA DEFINIRE" in core:
                self.warnings.append("Core Output contiene DA DEFINIRE")
        
        # 3. Controlla che le sezioni siano complete
        sections = ["STRATEGIA", "PIANO SOCIAL", "HASHTAG", "CTA", "TIMELINE"]
        for section in sections:
            if section not in output:
                self.warnings.append(f"Sezione '{section}' mancante")
        
        return len(self.errors) == 0, self.errors, self.warnings
    
    def get_report(self):
        return f"✅ Validazione superata" if not self.errors else f"❌ {len(self.errors)} errori"

"""
CreativeGate V1.8.1
Isola il Creative Lab e previene contaminatione del Core Output.
"""

import re

class CreativeGate:
    """Verifica che le idee creative restino nel Lab e non diventino fatti."""

    LAB_MARKER = "💡 CREATIVE LAB"

    def __init__(self, facts: dict):
        self.facts = {k: str(v).lower() for k, v in facts.items() if v}

    def split(self, plan: str):
        """Divide il piano in (core_output, creative_lab)."""
        idx = plan.find(self.LAB_MARKER)
        if idx == -1:
            return plan, ""
        core = plan[:idx].strip()
        lab = plan[idx:].strip()
        return core, lab

    def scan_contamination(self, core: str) -> list:
        """Trova entità/fatti creativi finiti nel Core Output."""
        flags = []
        core_l = core.lower()
        # Se il Core contiene marker creativi senza il Lab → contaminatione
        creative_terms = ["potremmo immaginare", "suggestivo", "idea creativa:",
                          "concept creativo", "🎨", "💡"]
        for t in creative_terms:
            if t in core_l:
                flags.append(f"Termine creativo '{t}' trovato nel Core Output")
        return flags

    def enforce(self, plan: str) -> tuple[str, list]:
        """Ritorna (piano_pulito, warnings)."""
        core, lab = self.split(plan)
        warnings = self.scan_contamination(core)
        if warnings:
            warn_block = "\n\n⚠️ CREATIVE GATE WARNING:\n" + "\n".join(f"- {w}" for w in warnings)
            core = core + warn_block
        return core + ("\n\n" + lab if lab else ""), warnings
